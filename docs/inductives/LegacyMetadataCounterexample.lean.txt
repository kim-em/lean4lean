import Lean4Lean.Verify.Inductive.Run.Formation

namespace InductiveBranchReview
open Lean Lean4Lean

-- The evidence used by AddConstants sees only the constant's name, type,
-- universe parameters and safety. It ignores the operational recursor metadata.
theorem recursor_metadata_is_unconstrained
    (safety : DefinitionSafety) (venv : VEnv) (rec : RecursorVal) (ci : VConstVal)
    (h : TrConstVal safety venv (.recInfo rec) ci)
    (arbitraryRules : List RecursorRule) (arbitraryNumParams : Nat)
    (arbitraryK : Bool) :
    TrConstVal safety venv
      (.recInfo { rec with rules := arbitraryRules, numParams := arbitraryNumParams, k := arbitraryK }) ci := h

#print axioms recursor_metadata_is_unconstrained
end InductiveBranchReview
